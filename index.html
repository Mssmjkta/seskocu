async function toggleMic() { if(isMicActive) { if(mic) mic.close(); isMicActive = false; document.getElementById('micBtn').classList.remove('active'); cancelAnimationFrame(detectionId); document.getElementById('tunerDisplay').innerText = "--"; document.getElementById('tunerDisplay').className = "tuner-display"; } else { if (Tone.context.state !== 'running') await Tone.start(); mic = new Tone.UserMedia(); try { // iOS MUSIC OPTIMIZATION: // Turning off echoCancellation and noiseSuppression // tells the iPhone NOT to use the "Phone Call" audio path. await mic.open({ echoCancellation: false, noiseSuppression: false, autoGainControl: true }); micGain = Tone.context.createGain(); const sliderVal = document.getElementById('micGainSlider').value; micGain.gain.value = parseFloat(sliderVal); // Filter: Lowered to 80Hz to allow more vocal resonance filterHighPass = new Tone.Filter(80, "highpass"); analyser = Tone.context.createAnalyser(); analyser.fftSize = 2048; mic.connect(filterHighPass); filterHighPass.connect(micGain); micGain.connect(analyser); isMicActive = true; document.getElementById('micBtn').classList.add('active'); document.getElementById('tunerDisplay').classList.add('listening'); updatePitch(); } catch(e) { console.error("Mic Error", e); alert("Microphone access denied or not found."); } } }
